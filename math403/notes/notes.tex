\section{Preliminaries}

\begin{theorem}
  \vocab{Well Ordering Principle} states that every nonempty set of positive integers contains a
  smallest number. 
\end{theorem}

\begin{definition}
  A nonzero integer $t$ is a \vocab{divisor} of an integer $s$ if there is an integer $u$ such that 
  $s = tu$ and we write $t | s$. A \vocab{prime} is a positive integer greater than $1$ whose only 
  positive divisors are $1$ and itself. We say an integer $s$ is a \vocab{multiple} of an integer $t$ if there is an integer $u$ such that 
  $s = tu$. 
\end{definition}

\begin{theorem}
  \vocab{Division Algorithm}: let $a$ and $b$ be integers with $b > 0$. Then there exist unique integers $q$ and $r$ with the property that 
  $a = bq + r$, where $0 \leq r < b$. 
  \begin{proof}
    Omitted. 
  \end{proof}
\end{theorem}

\begin{definition}
  The \vocab{greatest} common divisor of two integers is hte largest of all common divisors of $a$ and $b$. 
  We denote this by $\gcd(a, b)$. When $\gcd(a, b) = 1$, we say $a$ and $b$ are \vocab{relatively prime}. 
\end{definition}

\begin{theorem}
  For any nonzero integers $a$ and $b$, there exist integers $s$ and $t$ such that $\gcd(a, b) = as + bt$. Moreover, 
  $\gcd(a, b)$ is the smallest positive integer of the form $as + bt$. 
  \begin{proof}
    Omitted. 
  \end{proof}
\end{theorem}

\begin{corollary}
  Integers $a$ and $b$ are relatively prime if and only if there exist integers $s$ and $t$ 
  such that $as + bt = 1$.  
\end{corollary}

\begin{lemma}[Euclid's Lemma]
  If $p$ is a prime that divides $ab$, then $p$ divides $a$ or $p$ divides $b$. 
  \begin{proof}
    Omitted.
  \end{proof}
\end{lemma}

\begin{theorem}[Fundamental Theorem of Arithmetic]
  Every integer greater than $1$ is a prime or a product of primes. This product is unique, except for the 
  order in which factors appear. 
\end{theorem}

\begin{definition}
  The \vocab{least common multiple} of two nonzero integers $a$ and $b$ is the smallest postivie integer that is 
  a multiple of both $a$ and $b$. We denote this as $lcm(a, b)$. 
\end{definition}

\begin{definition}[Modular Arithmetic]
  When $a = qn + r$ where $q$ is the quotient and $r$ is the remainder upon dividing $a$ by $n$, 
  we write $a \mod n = r$. 
\end{definition}

\begin{definition}[Equivalence Relation]
  A \vocab{equivalence relation} on a set $S$ is a set $R$ of ordered pairs of elements of $S$ such that 
  \begin{enumerate}
    \item Reflexive: $(a, a) \in R$ for all $a \in S$
    \item Symmetric: $(a, b) \in R \implies (b, a) \in R$
    \item Transitive: $(a, b) \in R$ and $(b, c) \in R \implies (a,c) \in R$
  \end{enumerate}
  If $\sim$ is an equivalence relation on a set $S$ and $a \in S$, then the set $[a] = \{x \in S \ | \ x \sim a\}$ is called 
  the \vocab{equivlalence class} of $S$ containing $a$. 
\end{definition}

\begin{definition}
  A \vocab{partition} of a set $S$ is a collection of nonempty disjoint subsets of $S$ whose union is $S$. 
\end{definition}

\begin{theorem}
  The equivalence classes of an equivalence relation on a set $S$ constitute a partition of $S$. 
  Conversely, for any partition $P$ of $S$, there is an equivalence relation on $S$ whose 
  equivalence classes are the elements of $P$. 
  \begin{proof}
    Omitted
  \end{proof}
\end{theorem}

\begin{definition}
  A function $\phi$ from a set $A$ to set $B$ is a rule that assigns to each element of $a$ of $A$ 
  exactly one element $b$ of $B$. $A$ is the \vocab{domain}, $B$ is the \vocab{range}, and 
  the subset of $B$ comprising all images of elements of $A$ is called the \vocab{image of A under $\phi$}. 
\end{definition}

\begin{definition}
    Let $\phi: A \to B$ and $\varphi: B \to C$. The \vocab{composition $\varphi \phi$} is the mapping from 
    $A$ to $C$ defined by $\varphi(\phi(a))$ for all $a$ in $A$. 
\end{definition}

\begin{definition}
  A function $\phi$ from a set $A$ is \vocab{one-to-one} if for every $a_1, a_2 \in A$ then 
  \[\phi(a_1) = \phi(a_2) \implies a_1 = a_2\]
\end{definition}

\begin{definition}
  A function $\phi: A \to B$ is \vocab{onto} if each element of $B$ is the image of at least one element of $A$. 
  \[\forall b \in B, \ \exists a \in A \text{ s.t. } \phi(a) = b\]
\end{definition}

\begin{theorem}
  Given functions $\alpha: A \to B, \beta: B \to C, \gamma: C \to D$ then 
  \begin{itemize}
    \item Associativity: $\gamma(\beta \alpha) = (\gamma \beta)\alpha$
    \item If $\alpha, \beta$ are both 1-1, then $\beta\alpha$ is 1-1
    \item If $\alpha, \beta$ are onto, then $\beta, \alpha$ is onto
    \item If $\alpha$ is 1-1 and onto, then there is a function $\alpha^{-1}$ from $B$ onto $A$ 
    such that $(\alpha^{-1}\alpha)(a) = a \ \forall \ a \in A$ and $(\alpha\alpha^{-1})(b) = b \ \forall \ b \in B$
  \end{itemize}
\end{theorem}

\begin{definition}
  Let $G$ be a set. The \vocab{binary operation} on $G$ is a function that assigns each ordered pair of elements of $G$ an element of $G$. 
  This condition is called \vocab{closure}. 
\end{definition}

\begin{definition}
  Let $G$ be a set together with a binary operation that assigns to each ordered pair of elements of $G$ an 
  element $G$ dentoed by $ab$. We say $G$ is a \vocab{group} under this operation of the following three properties 
  are satisfied
  \begin{enumerate}
    \item Closure?
    \item Associativity: $(ab)c = a(bc) \ \forall \ a, b, c \in G$
    \item Identity: There is an element $e$ called the identity in $G$ such that $ae = ea = a \ \forall \ a \in G$
    \item Inverses: for each element $a \in G$, there is an element $b$ in $G$ such that $ab = ba = e$. 
  \end{enumerate}
\end{definition}

\begin{definition}
  If a group has the property that $ab = ba \ \forall a, b \in G$, then the group is \vocab{Abelian}. 
\end{definition}

\begin{example}
  \begin{itemize}
    \item $\ZZ$, $\QQ$, $\RR$ are all groups under ordinary addition. In each case, the identity is $0$ and the inverse of $a$ is $-a$. 
    \item The set of integers under ordinary multiplication is not a group, it fails the inverse property check. 
    \item The set $Z_n = \{0, 1, \ldots, n-1\}$ for $n \geq 1$ is a group under addition modulo $n$. 
    \item The set $\textbf{R}^*$ of nozero real numbers is a group under ordinary multplication with identity $1$ and inverse $\frac{1}{a}$ for any $a$. 
    \item The set 
    \[GL(2, \RR) = \left\{ \begin{bmatrix}
    a & b \\ c & d
    \end{bmatrix} a, b, c , d \in \RR, ad - bc \neq 0\right\}\]
    is a non-Abelian group under the operation of matrix multiplication. 
  \end{itemize}
\end{example}

\section{Finite Groups and Subgroups}

Groups with finitely many elements have interesting artihemtic properties. 

\begin{definition}[Order of a Group]
  The number of elements in a group (finite or infinite) is called its \vocab{order}. We will use 
  $|G|$ to denote the order of $G$.
\end{definition}

\begin{example}
  The group $\ZZ$ under addition has infinite order, whereas the grop $U(1, 3, 7, 9)$ under multiplication module $10$ has order $4$. 
\end{example}

\begin{definition}
  The order $g$ in a group $G$ is the smallest positive integer $n$ such that $g^n = e$. 
  In additive notation, this would be $ng = e$. If no such integer exists, we say that $g$ has infinite order. 
  The order of an element $g$ is denoted by $|g|$. 
\end{definition}

\begin{note}
The group $U(n)$ is the set of all positive integers less than and relatively prime to $n$. 
\end{note}

\begin{definition}[Subgroup]
  If a subset $H$ of a group is itself a group under the operation of $G$, we say that $H$ is a subgroup of $G$. 
  Subgroups still must have the same properties as groups (closure, associativity, identity, and inverses). 
\end{definition}

\subsection{Subgroup Tests}

\begin{theorem}[One-Step Subgroup Test]
  Let $G$ be a group and $H$ a nonempty subset of $G$. If $ab^{-1}$ is in $H$ wheneber $a, b \in H$, then $H$ is a subgroup 
  of $G$. In additive notation, if $a - b \in H$ whenever $a, b \in H$, then $H$ is a subgroup of $G$. 
  \begin{proof}
    Omitted.
  \end{proof} 
  There are actually four steps involved in applying the theorem. 
  \begin{enumerate}
    \item Identity the property $P$ that distinguishes the elements of $H$; that is, identify a defining condition. 
    \item Prove that the identity has property $P$; this verifies that $H$ is nonempty
    \item Assume that two elements $a$ and $b$ have property $P$. 
    \item Use the assumption that $a$ and $b$ have property $P$ to show that $ab^{-1}$ has property $P$. 
  \end{enumerate} 
\end{theorem}

\begin{theorem}[Two-Step Subgroup Test]
  Let $G$ be a group and let $H$ be a nonempty subset of $G$. If $ab \in H$ whenever $a, b \in H$
  and $a \in H \implies a^{-1} \in H$, then $H$ is a subgroup of $G$. 
  \begin{itemize}
    \item $H$ is closed under the operation, $a, b \in H \implies ab \in H$
    \item $H$ is closed under taking inverses, $a \in H \implies a^{-1} \in H$
  \end{itemize} 
  \begin{proof}
    Omitted.
  \end{proof}
  Same process as above, but we use the assumption that $a$ and $b$ have property $P$ to prove that $ab$ has property $P$ and $a^{-1}$ has property $b$. 
\end{theorem}

\begin{note}
How do you show that a subset of a group is \textit{not} a subgroup?
\begin{enumerate}
  \item Show that the identity is not in the set. 
  \item Exhibit an element of the set whose inverse is not in the set.
  \item Exhibit two elements of the set whose product is not in the set. 
\end{enumerate}
\end{note}

\begin{theorem}[Finite Subgroup Test]
  Let $H$ be a nonempty finite subset of a group $G$. If $H$ is closed under the operation of $G$, then $H$ is a subgroup of $G$. 
\end{theorem}

\begin{definition}
  For any element $a \in G$, we let $\langle a \rangle = \{a^n \ | \ n \in \ZZ\}$. In particular, observe
  that the exponents of $a$ include all negative integers as well as $0$ and the positive integers, 
  $a^0$ is defined to be the identity
\end{definition}

\begin{theorem}[$\langle a \rangle$ is a Subgroup]
  Let $G$ be a group, and let $a \in G$. Then $\langle a \rangle$ is a subgroup of $G$. 
  \begin{proof}
    Since $a \in \langle a \rangle$, $\langle a \rangle$ is not empty. Let $a^n, a^m \in \langle a \rangle$. Then 
    $a^n (a^m)^{-1} = a^{n-m} \in \langle a \rangle$; so by the One-Step Subgrpup Test, $\langle a \rangle$ is a subgroup of $G$. 
  \end{proof}
\end{theorem}

\begin{definition}[Center of a Group]
  The \vocab{center}, $Z(G)$, of a group $G$ is the subset of elements in $G$ that commute with every lement of $G$. 
  \[Z(G) = \{a \in G \ | \ ax = xa \ \forall \ x \in G\}\]
\end{definition}

\begin{theorem}[Center is a Subgroup]
  The center of a group $G$ is a subgroup of $G$.

\end{theorem}

\begin{definition}[Centralizer of $a$ in $G$]
  Let $a$ be a fixed element of $G$. The \vocab{centralizer} of $a$ in $G$, $C(a)$, is the set 
  of all elements in $G$ that commute with $a$
  \[C(a) = \{g \in G \ | \ ga = ag\}\]
\end{definition}

\begin{example}

\section{Cyclic Groups}

\begin{definition}
  A group $G$ is \vocab{cyclic} if there is an element $a \in G$ such that $G = \{a^n \ | \ n \in \ZZ\}$. Such an 
  element is called a \vocab{generator} of G. From now on, we may indicate that $G$ is a cyclic group generated by $a$ 
  by writing $G = \langle a \rangle$. 
\end{definition}

\begin{theorem}[Criterion for $a^i = a^j$]
  Let $G$ be a group, and let $a \in G$.
  \begin{itemize}
    \item If $a$ has infinite order, then $a^i = a^j$ if and only if $i = j$
    \item If $a$ has finite order, then $a^i = a^j$ if and only if $n$ divides $i - j$. 
  \end{itemize}
\end{theorem}

\begin{corollary}
  For any group element $a \in G$, $|a| = |\langle a \rangle|$. 
\end{corollary}

\begin{corollary}
  For any group element $a$, $a^k = e$ if and only if $|a|$ divides $k$. 
\end{corollary}

\begin{corollary}
  If $a, b$ belong to a finite group and $ab = ba$ then $|ab|$ divides $|a||b|$. 
\end{corollary}

\begin{theorem}
  Let $a \in G$ be such that $|a| = n$, and let $k$ be a positive integer. Then $\langle a^k \rangle = \langle a^{\gcd(n, k)}\rangle$ and 
  $|a^k| = \frac{n}{\gcd(n, k)}$. 
\end{theorem}

\begin{corollary}
  In a finite cyclic group, the order of an element divdes the order of the group.
\end{corollary}

\begin{corollary}
  Let $|a| = n$. Then $\langle a^i \rangle = \langle a^j \rangle$ if and only if $\gcd(n, i) = \gcd(n, j)$
  and $|a^i| = |a^j|$ if and only if $\gcd(n, i) = \gcd(n, j)$. 
\end{corollary}

\begin{corollary}
  Let $|a| = n$. Then $\langle a \rangle = \langle a^j\rangle$ if and oly if $\gcd(n, j) = 1$. 
\end{corollary}

\begin{corollary}
  An integer $k \in Z_n$ is a generator of $Z_n$ if and only if $\gcd(n, k) = 1$.  
\end{corollary}

\section{Permutation Groups}

At this point, we kow two families of finite Abelian groups $Z_n = \{0, 1, \ldots, n-1\}$
and $U(n)$ the set of all positive integers less than $n$ that are relatively prime with $n$. We also know two families 
of infinite non-Abelian groups, dihedral groups and matrix groups. 

\begin{definition}[Permutation of $A$, Permutation Group of $A$]
  A \vocab{permutation} of a set $A$ is a function from $A$ to $A$ that is both 1-1 and onto. A \vocab{permutation group} of a set $A$ 
  is a set of permutations of $A$ that forms a group under function composition. 
\end{definition}

\begin{definition}
  \vocab{Cycle notation} has theoretical advantages in that certain important properties of the permutation can be readily determined 
  when cycle notation is used. Consider the permutation 
  \[\alpha = \begin{bmatrix}
    1 & 2 & 3 & 4 & 5 & 6\\
    2 & 1 & 4 & 6 & 5 & 3
  \end{bmatrix}\]
  Instead, we have $\alpha= (12)(346)(5)$, where for example, $(346)$ is a 3-cycle. 
\end{definition}

\begin{theorem}[Properties of Disjoint Cycles]
  Every permutation of a finite set can be written as a cycle or as a product of disjoint cycles. 
\end{theorem}

\begin{theorem}[Disjoint Cycles Commute]
  If hte pair of cycles $\alpha = (a_1, a_2, \ldots, a_m)$ and $\beta = (b_1, b_2, \ldots, b_n)$ have no entries in common, then 
  $\alpha \beta = \beta \alpha$. 
\end{theorem}

\begin{theorem}[Order of a Permutation]
  The order of a permutation of a finite set written in disjoint cycle form is the least common multiple of the lengths of the cycles. 
\end{theorem}

\begin{theorem}[Product of 2-Cycles]
  Every permutation in $S_n, n > 1$ is a product of $2$-cycles. 
\end{theorem}

\begin{theorem}
  If $e = \beta_1 \beta_2 \ldots \beta_r$ where the $\beta$'s are 2-cycles, then $r$ is even. 
\end{theorem}

\begin{theorem}[Always Even or Always Odd]
  If a permutation $\alpha$ can be expressed as a product of an even (odd) number of 2-cycles, then every 
  decomposition of $\alpha$ into a product of 2-cucles must have an even (odd) number of 2-cycles. 
\end{theorem}

\begin{definition}[Even and Odd Permutations]
  A permutation that can be expressed as a product of an even (odd) number of 2-cycles is called an \vocab{even (odd)} permutation. 
\end{definition}

\begin{theorem}
  The set of even permutations in $S_n$ forms a subgroup of $S_n$. 
\end{theorem}

\begin{definition}
  The group of even permutations of $n$ symbols is denoted by $A_n$ and is called the \vocab{alternating group of degree $n$}.
\end{definition}

\begin{theorem}
  For $n > 1, A_n $ has order $n! / 2$. 
\end{theorem}

\section{Isomorphisms}

\begin{definition}
  An \vocab{isomorphism $\phi$} from a group $G$ to a group $G'$ is a 1-1 and onto mapping from $G$ to $G'$ that 
  preserves the group operation. That is,
  \begin{align*}
    \phi(ab) = \phi(a)\phi(b) \ \forall \ a, b \in G
  \end{align*}
  If there is an isomorphism from $G$ onto $G'$, then we say that $G$ and $G'$ are isomorphic and we write 
  $G \approx G'$. 
\end{definition}

\begin{center}
  \includegraphics[width=6cm]{ops-table.png}
  \includegraphics[width=14cm]{ops-table2.png}
\end{center}

\begin{note}
  To prove that a group $G$ is isomorphic to a group $G'$, 
  \begin{enumerate}
    \item Define a function $\phi: G \to G'$
    \item 1-1: prove that $\phi$ is 1-1, so $\phi(a) = \phi(b) \implies a = b$
    \item Onto: prove that $\phi$ is onto, so for any $g' \in G'$, find a $g \in G$ such that $\phi(g) = g'$
    \item Prove that $\phi$ is operation-preserving; that is show that $\phi(ab) = \phi(a)\phi(v) \ \forall \ a, b \in G$. 
  \end{enumerate}
\end{note}

\begin{theorem}[Properties of Isomorphisms Acting on Elements]
  \hfill

  \begin{enumerate}
    \item $\phi$ carries the identity of $G$ to the identity of $G'$
    \item For every integer $n$ and for every $a \in G$, then $\phi(a^n) = [\phi(a)]^n$, and the 
    additive form is $\phi(na) = n\phi(a)$
    \item For any $a, b \in G$, $a$ and $b$ commute if and only if $\phi(a)$ and $\phi(b)$ commute 
    \item $G = \langle a \rangle$ iff $G' = \langle \phi(a) \rangle$
    \item $|a| = |\phi(a)| \ \forall \ a \in G$ --- isomorphisms preserve orders 
    \item If $G$ is finite, then $G$ and $G'$ have exactly the same number of ekements of every order.
  \end{enumerate}
\end{theorem}

\begin{theorem}[Properties of Isomorphisms Acting on Groups]
  \hfill

  \begin{enumerate}
    \item $\phi^{-1}$ is an isomorphism from $G'$ onto $G$
    \item $G$ is Abelian if and only if $G'$ is Abelian
    \item $G$ is cyclic if and only if $G'$ is cyclic 
    \item If $K$ is a subgroup of $G$, then $\phi(K) = \{\phi(k) \ | \ k \in K\}$ is a subgroup of $G'$ 
    \item If $K'$ is a subgroup of $G'$, then $\phi^{-1}(K') = \{g \in G \ | \ \phi(g) \in K'\}$ is a subgroup of $G$ 
    \item $\phi(Z(G)) = Z(G')$
  \end{enumerate}
\end{theorem}

\subsection{Automorphisms}

\begin{definition}
  An isomorphism from a group $G$ onto itself is called an \vocab{automorphism} of $G$ .
\end{definition}

\begin{definition}
  Let $G$ be a group, and let $a \in G$. The function $\phi_a$ defined by $\phi_a(x) = axa^{-1}$ for all $x \in G$ is called the \vocab{inner automorphism of $G$ induced by a}.
\end{definition}

\begin{theorem}
  The set of automorphisms of a group and the set of inner automorphisms of a group are both groups under the operation of function composition.
\end{theorem}

\begin{theorem}[Cayley's Theorem]
  Every group is isomorphic to a group of permutations. 
\end{theorem}

\section{Cosets and Lagrange's Theorem}

\begin{definition}
  Let $G$ be a group and let $H$ be a nonempty subset of $G$. For any $a \in G$, the set 
  \[aH = \{ah \ | \ h \ in H\} \ \ \ \ \ \ \{Ha = \{ha \ | \ h \in H\}\}\]
  When $H$ is a subgroup of $G$, the set $aH$ is called the \vocab{left coset of H in G containing a}, and similarly for 
  \vocab{right coset}. In this case, the element $a$ is the \vocab{coset representative}. 
\end{definition}

\begin{lemma}
  Let $H$ be a subgroup of $G$, and let $a, b \in G$. Then 
  \begin{enumerate}
    \item $a \in aH$ 
    \item $aH = H$ if and only if $a \in H$
    \item $(ab)H = a(bH)$ if and only if $H(ab) = (Ha)b$
    \item $aH = bH$ if and oly if $a \in bH$
    \item $aH = bH$ or $aH \cap bH = \emptyset$
    \item $aH = bH$ if and only if $a^{-1}b \in H$
    \item $|aH| = |bH|$
  \end{enumerate}
\end{lemma}

\begin{theorem}[Lagrange's Theorem]
  If $G$ is a finite group and $H$ is a subgroup of $G$, then $|H|$ divides 
  $|G|$. Moreover, the number of distinct left (right) cosets of $H$ in $G$ is 
  $|G| / |H|$. 
\end{theorem}

\begin{corollary}
  In a finite group, the order of each element of the group divides the order of the group. 
\end{corollary}

\begin{corollary}
  Every group of prime order is isomorphic to $Z_p$. 
\end{corollary}

\begin{corollary}
  Let $G$ be a finite group, and let $a \in G$. Then, $a^{G} = e$. 
\end{corollary}

\begin{corollary}
  For every integer $a$ and every prime $p$, $a^p \mod p = a \mod p$. 
\end{corollary}

\begin{theorem}
  For two finite subgroups $H$ and $K$ of a group, define the set $HK = \{hk \ | \ h \in H, k \in K\}$. Then
  \[|HK| = |H||K| = |H||K| / |H \cap K|\]
\end{theorem}

\begin{theorem}
  Let $G$ be a group of order $2p$, where $p$ is a prime greater than $2$. Then $G$ is isomorphic to $Z_{2p}$ or $D_p$. 
\end{theorem}

\section{External Direct Products}

\begin{definition}[External Direct Product]
  Let $G_1, \ldots, G_n$ be a finite collection of groups. The external direct product of such 
  groups written as $G_1 \oplus G_2 \oplus \cdots \oplus G_n$ is the set of all $n$-tuples for which the $i$-th component is an element of $G_i$, and 
  the operation is componentwise.
\end{definition}

\begin{example}
  Indeed, $\RR^2 = \RR \oplus \RR$ and the operation is componentwise/vector addition. Another example is 
  \[Z_2 \oplus Z_3 = \{(0, 0), (0, 1), (0, 2), (1, 0), (1,1), (1,2)\}\]
\end{example}

\begin{theorem}
  The order of an element in a direct product of a finite number of finite groups is the LCM of the orders of the components of the element. 
  \[|(g_1, g_2, \ldots, g_n)| = \text{lcm}(|g_1|, |g_2|, \ldots, |g_n|)\]
\end{theorem}

\begin{theorem}[Criterion for $G \oplus H$ to be Cyclic]
  Let $G$ and $H$ be finite cyclic groups. Then $G \oplus H$ is cyclic if and only if $|G|$ and $|H|$ are relatively prime. 
\end{theorem}

\begin{corollary}[Criterion for $G_1 \oplus G_2 \oplus \ldots G_n$ to be cylic]
  An external direct product $G_1 \oplus G_2 \oplus \cdots \oplus G_n$ of a finite number of finite 
  cylic groups is clyclif if and only if $|G_1|$ and $|G_j|$ are relatively prime when $i \neq j$. 
\end{corollary}

\begin{corollary}
  Let $m = n_1n_2\cdots n_k$. Then $Z_m$ is isomorphic to $Z_{n_1} \oplus Z_{n_2} \oplus \cdots \oplus Z_{n_k}$ if and only if 
  $n_i$ and $n_j$ are relatively prime when $i \neq j$. 
\end{corollary}

\begin{example}
  For example, we have 
  \[Z_2 \oplus Z_2 \oplus Z_3 \oplus Z_5 \approx Z_2 \oplus Z_6 \oplus Z_5 \approx Z_2 \oplus Z_{30}\]
  but note that $Z_2 \oplus Z_{30} \not\approx Z_{60}$
\end{example}

\begin{theorem}[$U(n)$ as an External Direct Product]
  Suppose $s$ and $t$ are relatively prime. Then $U(st)$ is isomorphic to the external direct product of 
  $U(s)$ and $U(t)$. In short, 
  \[U(st) \approx U(s) \oplus U(t)\]
  Moreover, $U_s(st)$ is isomorphic to $U(t)$ and $U_t(st)$ is isomorphic to $U(s)$. 
\end{theorem}

\begin{corollary}
  Let $m = n_1n_2 \cdots n_k$ where $\gcd(n_i, n_j) = 1 \ \forall \ i \neq j$. Then 
  \[U(m) \approx U(n_1) \oplus U(n_2) \oplus \cdots \oplus U(n_k)\]
\end{corollary}

\section{Normal Subgroups and Factor Groups}

\begin{definition}[Normal Subgroup]
  A subgroup $H$ of a group $G$ is called a \vocab{normal subgroup} of $G$ if $aH = Ha \ \forall \ a \in G$. We denote this 
  by $H \lhd G$. 
\end{definition}

\begin{theorem}[Normal Subgroup Test]
  A subgroup $H$ of $G$ is normal in $G$ if and only if $xHx^{-1} \subseteq H \ \forall \ x \in G$. 
\end{theorem}

\begin{theorem}[Factor Groups]
  Let $G$ be a group and let $H$ be a normaml subgroup of $G$. The set $G \slash H = \{aH \ | \ a \in G\}$ is a group 
  under the operation $(aH)(bH) = abH$.
\end{theorem}

\begin{theorem}[$G \slash Z$ Theorem]
  Let $G$ be a group and let $Z(G)$ be the center of $G$. If $G \slash Z(G)$ is cyclic, then $G$ is Abelian.
\end{theorem}

\begin{theorem}[$G \slash Z(G) \approx Inn(G)$]
  For any group $G$, $G \slash Z(G) \approx Inn(G)$
\end{theorem}
